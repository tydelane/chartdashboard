class ApiError(Exception):
    def __init__(self, status, message, detail=None):
        self.status = int(status)
        self.message = message
        self.detail = detail
        Exception.__init__(self, message)

def _set_http_status(request, status):
    try:
        response = request.get("servletResponse")
        if response is not None:
            response.setStatus(int(status))
            response.setHeader("Cache-Control", "no-store, no-cache, must-revalidate")
            response.setHeader("Pragma", "no-cache")
    except Exception:
        pass

def _json_ok(request, data):
    _set_http_status(request, 200)
    return {"json": {"ok": True, "data": data}}

def _json_error(request, status, message, request_id=None):
    _set_http_status(request, status)
    payload = {
        "ok": False,
        "error": {
            "status": int(status),
            "message": message
        }
    }
    if request_id:
        payload["error"]["requestId"] = request_id
    return {"json": payload}

def _log_exception(where, exc):
    try:
        LOG.error("%s: %s\n%s" % (where, exc, traceback.format_exc()))
    except Exception:
        pass

def _as_text(value):
    if value is None:
        return ""
    try:
        return unicode(value)
    except Exception:
        return str(value)

def _utf8_length(value):
    return len(_as_text(value).encode("utf-8"))

def _json_bytes(value):
    try:
        return _utf8_length(system.util.jsonEncode(value))
    except Exception:
        return MAX_JSON_REQUEST_BYTES + 1

def _bool(value, default=False):
    if isinstance(value, bool):
        return value
    if value is None:
        return bool(default)
    return _as_text(value).strip().lower() in ("1", "true", "yes", "on")

def _int(value, label, minimum=None, maximum=None):
    try:
        # bool is not accepted as an integer configuration field.
        if isinstance(value, bool):
            raise ValueError()
        result = int(value)
    except Exception:
        raise ApiError(400, "%s must be an integer." % label)
    if minimum is not None and result < minimum:
        raise ApiError(400, "%s must be at least %s." % (label, minimum))
    if maximum is not None and result > maximum:
        raise ApiError(400, "%s must be at most %s." % (label, maximum))
    return result

def _number_or_none(value, label, minimum=None, maximum=None):
    if value is None or value == "":
        return None
    try:
        result = float(value)
    except Exception:
        raise ApiError(400, "%s must be numeric." % label)
    if minimum is not None and result < minimum:
        raise ApiError(400, "%s is below the allowed minimum." % label)
    if maximum is not None and result > maximum:
        raise ApiError(400, "%s is above the allowed maximum." % label)
    return result

def _safe_text(value, label, maximum, allow_empty=True):
    text = _as_text(value).strip()
    if not allow_empty and not text:
        raise ApiError(400, "%s is required." % label)
    if len(text) > maximum:
        raise ApiError(400, "%s is too long." % label)
    if _SAFE_TEXT_FORBIDDEN_RE.search(text):
        raise ApiError(400, "%s contains unsupported characters." % label)
    # Explicitly reject script-ish payloads; frontend only renders text with
    # textContent, but the stored JSON should never become a code container.
    lowered = text.lower()
    if "javascript:" in lowered or "<script" in lowered or "function(" in lowered:
        raise ApiError(400, "%s contains unsupported content." % label)
    return text

def _safe_color(value, label, default=None):
    if value is None or value == "":
        return default
    text = _as_text(value).strip()
    if not _COLOR_RE.match(text):
        raise ApiError(400, "%s must be a six-digit hex color such as #5CA7FF." % label)
    return text.upper()

def _require_dict(value, label):
    if isinstance(value, dict):
        return value
    # Web Dev/Jython can hand JSON objects back as Java Map-like objects.
    try:
        converted = dict(value)
        if isinstance(converted, dict):
            return converted
    except Exception:
        pass
    raise ApiError(400, "%s must be an object." % label)

def _require_list(value, label):
    if isinstance(value, (list, tuple)):
        return list(value)
    # JSON arrays can arrive as Java List-like values in some gateway builds.
    try:
        if value is not None and not isinstance(value, (str, unicode)):
            return [item for item in value]
    except Exception:
        pass
    raise ApiError(400, "%s must be a list." % label)

def _assert_known_keys(value, known_keys, label):
    unknown = [key for key in value.keys() if key not in known_keys]
    if unknown:
        raise ApiError(400, "%s contains unsupported setting(s): %s." % (label, ", ".join([_as_text(k) for k in unknown])))

def _validate_identifier(value, label):
    text = _as_text(value)
    if not _IDENTIFIER_RE.match(text):
        raise RuntimeError("%s must be a SQL identifier containing only letters, numbers, and underscores." % label)
    return text

def _java_list(value):
    if value is None:
        return []
    try:
        return [item for item in value]
    except Exception:
        return [value]

def _call_user_method(user, names):
    for name in names:
        try:
            method = getattr(user, name)
            return method()
        except Exception:
            pass
    return None

def _authenticated_identity(session):
    """
    Uses ONLY Web Dev's server-populated session['user'].
    The browser never supplies a username, display name, role, or permission.
    """
    if session is None or "user" not in session or session.get("user") is None:
        raise ApiError(401, "Authentication is required. Configure Web Dev Require Authentication for this resource.")
    user = session.get("user")

    username = _call_user_method(user, ("getUsername", "getUserName", "getName"))
    if username is None:
        try:
            username = user.get("username")
        except Exception:
            username = None
    username = _safe_text(username, "Authenticated username", 255, allow_empty=False)

    display_name = _call_user_method(user, ("getDisplayName", "getFullName", "getName"))
    if not display_name:
        display_name = username
    display_name = _safe_text(display_name, "Authenticated display name", 255, allow_empty=False)

    role_values = _java_list(_call_user_method(user, ("getRoles", "getRoleNames")))
    roles = []
    for role in role_values:
        text = _as_text(role).strip()
        if text:
            roles.append(text)

    # Newer user objects may expose security levels. Web Dev documented auth is
    # User Source / role based, so this is an optional compatibility attempt,
    # not a replacement for the resource's Require Authentication setting.
    security_values = _java_list(_call_user_method(user, ("getSecurityLevels",)))
    security_keys = list(roles)
    for level in security_values:
        try:
            security_keys.append(_as_text(level.getPath()))
        except Exception:
            security_keys.append(_as_text(level))
    security_keys = sorted(list(set([x for x in security_keys if x])))

    return {
        "username": username,
        "displayName": display_name,
        "roles": sorted(list(set(roles))),
        "securityKeys": security_keys
    }

def _matches_any(identity, allowed):
    if not allowed:
        return False
    user_keys = set(identity.get("securityKeys", []))
    for key in allowed:
        if _as_text(key) in user_keys:
            return True
    return False

def _permissions(identity):
    can_view = _matches_any(identity, REQUIRED_VIEW_ROLES_OR_SECURITY_LEVELS)
    can_edit = _matches_any(identity, REQUIRED_EDIT_ROLES_OR_SECURITY_LEVELS)
    is_admin = _matches_any(identity, ADMIN_ROLES_OR_SECURITY_LEVELS)
    # Administration is also explicitly allowed to view/edit only if listed in
    # the respective arrays. This avoids accidental privilege escalation.
    return {
        "canView": bool(can_view),
        "canEdit": bool(can_edit),
        "isAdmin": bool(is_admin)
    }

def _require_view(identity):
    permissions = _permissions(identity)
    if not permissions["canView"]:
        raise ApiError(403, "You are authenticated but are not authorized to view this dashboard.")
    return permissions

def _require_edit(identity):
    permissions = _require_view(identity)
    if not permissions["canEdit"]:
        raise ApiError(403, "You are not authorized to create or edit dashboards.")
    return permissions

def _request_scheme_is_https(request):
    try:
        return _as_text(request.get("scheme")).lower() == "https"
    except Exception:
        return False

def _enforce_transport(request):
    if ENFORCE_HTTPS_IN_CODE and not _request_scheme_is_https(request):
        raise ApiError(403, "HTTPS is required for this dashboard.")

def _ensure_csrf(session, action, request):
    # Bootstrap is intentionally the only action allowed before a token exists.
    # It only returns the authenticated user's own non-secret capabilities.
    if action == "auth.bootstrap":
        if not session.get("pmdCsrf"):
            session["pmdCsrf"] = _as_text(UUID.randomUUID())
        return
    expected = _as_text(session.get("pmdCsrf"))
    actual = ""
    try:
        headers = request.get("headers", {})
        actual = _as_text(headers.get("X-PMD-CSRF") or headers.get("x-pmd-csrf"))
    except Exception:
        actual = ""
    if not expected or actual != expected:
        raise ApiError(403, "Your dashboard session token is missing or expired. Refresh the page and try again.")

def _dashboard_summary_from_row(row):
    return {
        "dashboardId": _as_text(row["DashboardId"]),
        "ownerUsername": _as_text(row["OwnerUsername"]),
        "ownerDisplayName": _as_text(row["OwnerDisplayName"]),
        "name": _as_text(row["DashboardName"]),
        "description": _as_text(row["DashboardDescription"]),
        "isShared": bool(row["IsShared"]),
        "configSchemaVersion": int(row["ConfigSchemaVersion"]),
        "createdAt": _millis(row["CreatedAt"]),
        "updatedAt": _millis(row["UpdatedAt"]),
        "rowVersion": _as_text(row["RowVersionToken"])
    }

def _dashboard_select_sql(where_sql):
    return (
        "SELECT CONVERT(VARCHAR(36), [DashboardId]) AS [DashboardId], "
        "[OwnerUsername], [OwnerDisplayName], [DashboardName], [DashboardDescription], "
        "[IsShared], [ConfigSchemaVersion], [ConfigJson], [CreatedAt], [UpdatedAt], "
        "CONVERT(VARCHAR(34), [RowVersion], 1) AS [RowVersionToken] "
        "FROM %s WHERE %s" % (_DASHBOARD_TABLE_SQL, where_sql)
    )

def _find_dashboard(dashboard_id):
    dashboard_id = _safe_text(dashboard_id, "Dashboard ID", 64, allow_empty=False)
    sql = _dashboard_select_sql("[DashboardId] = CONVERT(UNIQUEIDENTIFIER, ?)")
    result = _rows(_query(sql, [dashboard_id]))
    if not result:
        raise ApiError(404, "Dashboard not found.")
    row = result[0]
    try:
        config = system.util.jsonDecode(_as_text(row["ConfigJson"]))
    except Exception as exc:
        _log_exception("Stored dashboard JSON is invalid", exc)
        raise ApiError(500, "A saved dashboard configuration is invalid. Contact an administrator.")
    return row, config

def _dashboard_access(row, identity, permissions):
    owner = _as_text(row["OwnerUsername"]) == identity["username"]
    shared = bool(row["IsShared"])
    can_read = owner or shared or (permissions["isAdmin"] and ALLOW_ADMIN_EDIT_ANY_DASHBOARD)
    can_edit_original = owner or (permissions["isAdmin"] and ALLOW_ADMIN_EDIT_ANY_DASHBOARD)
    return {
        "isOwner": owner,
        "canRead": can_read,
        "canEdit": can_edit_original,
        "isShared": shared
    }

def _dashboard_payload(row, config, identity, permissions):
    access = _dashboard_access(row, identity, permissions)
    if not access["canRead"]:
        raise ApiError(403, "You are not authorized to open this dashboard.")
    result = _dashboard_summary_from_row(row)
    result["config"] = config
    result["isOwner"] = access["isOwner"]
    result["canEdit"] = access["canEdit"]
    return result

def _count_owner_dashboards(username):
    sql = "SELECT COUNT(1) AS [Count] FROM %s WHERE [OwnerUsername] = ?" % _DASHBOARD_TABLE_SQL
    rows = _rows(_query(sql, [username]))
    return int(rows[0]["Count"]) if rows else 0

def _unique_dashboard_name(owner_username, requested_name):
    base = _safe_text(requested_name, "Dashboard name", MAX_DASHBOARD_NAME_LENGTH, allow_empty=False)
    candidate = base
    suffix = 2
    while True:
        sql = "SELECT COUNT(1) AS [Count] FROM %s WHERE [OwnerUsername] = ? AND [DashboardName] = ?" % _DASHBOARD_TABLE_SQL
        exists = int(_rows(_query(sql, [owner_username, candidate]))[0]["Count"]) > 0
        if not exists:
            return candidate
        suffix_text = " (%d)" % suffix
        candidate = base[:MAX_DASHBOARD_NAME_LENGTH - len(suffix_text)] + suffix_text
        suffix += 1

def _allowed_root_for_path(path):
    for root in ALLOWED_TAG_ROOTS:
        configured_root = _as_text(root.get("root")).rstrip("/")
        if path == configured_root or path.startswith(configured_root + "/"):
            return root
    return None

def _tag_provider(path):
    match = _TAG_PROVIDER_RE.match(path)
    return match.group(1) if match else None

def _validate_tag_path(value, label="Tag path"):
    path = _safe_text(value, label, 1024, allow_empty=False)
    if "\n" in path or "\r" in path or "\x00" in path:
        raise ApiError(400, "%s is invalid." % label)
    provider = _tag_provider(path)
    if provider is None:
        raise ApiError(400, "%s must include an explicit provider, for example [default]Area/Tag." % label)
    if provider not in ALLOWED_TAG_PROVIDERS:
        raise ApiError(403, "The tag provider is not approved for this dashboard.")
    if _allowed_root_for_path(path) is None:
        raise ApiError(403, "The tag path is outside the approved roots.")
    return path

def _path_depth(root_path, candidate_path):
    if candidate_path == root_path:
        return 0
    relative = candidate_path[len(root_path):].strip("/")
    return 0 if not relative else len(relative.split("/"))

def _tag_kind_from_datatype(data_type):
    text = _as_text(data_type).lower()
    if "bool" in text:
        return "boolean"
    # The browser may change an integer analog tag to state mode explicitly.
    return "analog"

def _browse_node_metadata(path):
    """
    Direct parent browse gives practical live metadata without allowing any
    write/configure operation. If direct browse metadata cannot be found, a
    safe conservative object is returned and history eligibility remains False.
    """
    path = _validate_tag_path(path)
    slash = path.rfind("/")
    parent = path[:slash] if slash > path.rfind("]") else path[:path.find("]") + 1]
    try:
        results = system.tag.browse(parent, {"maxResults": TAG_BROWSER_MAX_RESULTS}).getResults()
        for node in results:
            full_path = _as_text(node.get("fullPath"))
            if full_path == path:
                data_type = _as_text(node.get("dataType") or "")
                tag_type = _as_text(node.get("tagType") or "")
                return {
                    "path": path,
                    "name": _as_text(node.get("name") or path.split("/")[-1]),
                    "dataType": data_type,
                    "tagType": tag_type,
                    "hasChildren": bool(node.get("hasChildren")),
                    "kind": _tag_kind_from_datatype(data_type)
                }
    except Exception as exc:
        _log_exception("Tag metadata browse failed", exc)
    return {
        "path": path,
        "name": path.split("/")[-1],
        "dataType": "",
        "tagType": "",
        "hasChildren": False,
        "kind": "analog"
    }

def _history_enabled_for_tag(path):
    if not REQUIRE_HISTORY_ENABLED_TAGS:
        return True
    # Tag configuration commonly omits properties that are inherited/defaulted.
    # Therefore only an explicit historyEnabled=False excludes a tag here.
    # The historian request is still server-side and returns an error when a
    # provider cannot supply history, rather than silently fabricating values.
    try:
        configs = system.tag.getConfiguration(path, False)
        if configs:
            cfg = configs[0]
            if "historyEnabled" in cfg:
                return bool(cfg.get("historyEnabled"))
    except Exception as exc:
        _log_exception("History-enabled metadata check failed", exc)
    return True

def _tag_metadata(path):
    meta = _browse_node_metadata(path)
    meta["historyEligible"] = _history_enabled_for_tag(meta["path"])
    return meta

def _validate_tag_paths(paths, maximum, label):
    items = _require_list(paths, label)
    if not items:
        raise ApiError(400, "%s cannot be empty." % label)
    if len(items) > maximum:
        raise ApiError(400, "%s exceeds the configured limit." % label)
    output = []
    seen = set()
    for item in items:
        path = _validate_tag_path(item, "Tag path")
        if path not in seen:
            seen.add(path)
            output.append(path)
    if not output:
        raise ApiError(400, "%s cannot be empty." % label)
    return output

def _validate_state_labels(value, label):
    if value is None:
        return {}
    labels = _require_dict(value, label)
    if len(labels) > 64:
        raise ApiError(400, "%s has too many state labels." % label)
    clean = {}
    for raw_key, raw_text in labels.items():
        key = _safe_text(raw_key, "%s key" % label, 32, allow_empty=False)

def _validate_tag_config(value, label, require_axis=False, axis_count=None):
    """
    Validates a tag instance stored inside ConfigJson. Every path is rechecked
    against ALLOWED_TAG_ROOTS; display/type metadata remains informational.
    """
    tag = _require_dict(value, label)
    _assert_known_keys(tag, (
        "id", "path", "label", "displayName", "datatype", "historyEligible",
        "renderMode", "axisIndex", "color", "lineType", "lineWidth",
        "visible", "aggregation", "units", "precision"
    ), label)
    tag_id = _safe_text(tag.get("id"), "%s id" % label, 96, allow_empty=False)
    if not _WIDGET_ID_RE.match(tag_id):
        raise ApiError(400, "%s id is invalid." % label)
    path = _validate_tag_path(tag.get("path"), "%s path" % label)
    render_mode = _as_text(tag.get("renderMode") or "analog").lower()
    if render_mode not in _ALLOWED_RENDER_KINDS:
        raise ApiError(400, "%s uses an unsupported render mode." % label)
    line_type = _as_text(tag.get("lineType") or "solid").lower()
    if line_type not in _ALLOWED_LINE_TYPES:
        raise ApiError(400, "%s has an unsupported line type." % label)
    aggregation = _as_text(tag.get("aggregation") or "Raw")
    if aggregation != "Raw" and aggregation not in ALLOWED_HISTORY_AGGREGATION_MODES:
        raise ApiError(400, "%s has an unsupported aggregation." % label)
    axis_index = _int(tag.get("axisIndex", 0), "%s axis index" % label, 0, max(0, (axis_count or 1) - 1)) if require_axis else 0
    return {
        "id": tag_id,
        "path": path,
        "label": _safe_text(tag.get("label") or path.split("/")[-1], "%s label" % label, MAX_TAG_LABEL_LENGTH, allow_empty=False),
        "displayName": _safe_text(tag.get("displayName") or "", "%s display name" % label, MAX_TAG_LABEL_LENGTH, allow_empty=True),
        "datatype": _safe_text(tag.get("datatype") or "", "%s datatype" % label, 80, allow_empty=True),
        "historyEligible": _bool(tag.get("historyEligible"), True),
        "renderMode": render_mode,
        "axisIndex": axis_index,
        "color": _safe_color(tag.get("color"), "%s color" % label, "#5CA7FF"),
        "lineType": line_type,
        "lineWidth": _int(tag.get("lineWidth", 2), "%s line width" % label, 1, 8),
        "visible": _bool(tag.get("visible"), True),
        "aggregation": aggregation,
        "units": _safe_text(tag.get("units") or "", "%s units" % label, 32, allow_empty=True),
        "precision": _int(tag.get("precision", DEFAULT_AXIS_PRECISION), "%s precision" % label, 0, 8)
    }

def _validate_optional_tag(value, label, require_axis=False, axis_count=None):
    if value is None:
        return None
    return _validate_tag_config(value, label, require_axis, axis_count)

def _validate_axis(value, index):
    axis = _require_dict(value, "Axis")
    _assert_known_keys(axis, (
        "id", "title", "side", "visible", "offset", "min", "max",
        "units", "precision"
    ), "Axis")
    axis_id = _safe_text(axis.get("id"), "Axis id", 96, allow_empty=False)
    if not _WIDGET_ID_RE.match(axis_id):

    side = _as_text(axis.get("side") or "left").lower()
    if side not in _ALLOWED_AXIS_SIDES:

    minimum = _number_or_none(axis.get("min"), "Axis minimum")
    maximum = _number_or_none(axis.get("max"), "Axis maximum")
    if minimum is not None and maximum is not None and minimum >= maximum:
        raise ApiError(400, "Axis minimum must be below maximum.")
    return {
        "id": axis_id,
        "title": _safe_text(axis.get("title") or ("Axis %d" % (index + 1)), "Axis title", 80, allow_empty=False),
        "side": side,
        "visible": _bool(axis.get("visible"), True),
        "offset": _int(axis.get("offset", 0), "Axis offset", 0, 400),
        "min": minimum,
        "max": maximum,
        "units": _safe_text(axis.get("units") or "", "Axis units", 32, allow_empty=True),
        "precision": _int(axis.get("precision", DEFAULT_AXIS_PRECISION), "Axis precision", 0, 8)
    }

def _validate_style(value, label):

        "accentColor": accent
    }

def _validate_widget_config(widget_type, config):
    config = _require_dict(config, "Widget config")
    allowed_by_type = {
        "trend": ("axes", "tags", "style"),
        "referenceValue": ("tag", "units", "precision", "accent", "showDelta"),
        "digitalStatus": ("tag", "onText", "offText", "trueColor", "falseColor"),
        "stateTimeline": ("tag", "states"),
        "referenceTable": ("tags",),
        "rangeStatistic": ("tag", "metric", "stateValue", "units", "precision"),
        "stateDuration": ("tag", "states"),
        "recentEvents": ("tags", "maxRows")
    }
    _assert_known_keys(config, allowed_by_type[widget_type], "Widget config")

    if widget_type == "trend":
        axes = [_validate_axis(item, index) for index, item in enumerate(_require_list(config.get("axes", []), "Trend axes"))]
        if not axes:
            axes = [_default_axis()]
        if len(axes) > MAX_AXES_PER_TREND:
            raise ApiError(400, "Trend Chart has too many axes.")
        axis_ids = set([axis["id"] for axis in axes])
        if len(axis_ids) != len(axes):
            raise ApiError(400, "Trend Chart axis IDs must be unique.")
        tags = [_validate_tag_config(item, "Trend pen", True, len(axes)) for item in _require_list(config.get("tags", []), "Trend pens")]
        if len(tags) > MAX_TREND_PENS_PER_WIDGET:
            raise ApiError(400, "Trend Chart has too many pens.")
        return {"axes": axes, "tags": tags, "style": _validate_style(config.get("style"), "Trend style")}

    if widget_type == "referenceValue":
        return {
            "tag": _validate_optional_tag(config.get("tag"), "Reference Value tag"),
            "units": _safe_text(config.get("units") or "", "Reference Value units", 32, allow_empty=True),
            "precision": _int(config.get("precision", DEFAULT_AXIS_PRECISION), "Reference Value precision", 0, 8),
            "accent": _safe_color(config.get("accent"), "Reference Value accent", "#5CA7FF"),
            "showDelta": _bool(config.get("showDelta"), False)
        }

    if widget_type == "digitalStatus":
        return {
            "tag": _validate_optional_tag(config.get("tag"), "Digital Status tag"),
            "onText": _safe_text(config.get("onText") or "On", "On text", 40, allow_empty=False),
            "offText": _safe_text(config.get("offText") or "Off", "Off text", 40, allow_empty=False),
            "trueColor": _safe_color(config.get("trueColor"), "True color", "#4FD1A5"),
            "falseColor": _safe_color(config.get("falseColor"), "False color", "#FB7185")
        }

    if widget_type == "stateTimeline":
        return {"tag": _validate_optional_tag(config.get("tag"), "State Timeline tag"), "states": _validate_state_labels(config.get("states"), "Timeline states")}

    if widget_type == "referenceTable":
        tags = [_validate_tag_config(item, "Reference Table tag") for item in _require_list(config.get("tags", []), "Reference Table tags")]
        if len(tags) > MAX_TABLE_TAGS:
            raise ApiError(400, "Reference-value Table has too many tags.")
        return {"tags": tags}

    if widget_type == "rangeStatistic":
        metric = _as_text(config.get("metric") or "average")
        valid_metrics = ("average", "minimum", "maximum", "sum", "change", "range", "durationOn", "durationOff", "percentOn", "transitionCount", "stateDuration", "statePercent", "stateChangeCount")
        if metric not in valid_metrics:
            raise ApiError(400, "Range Statistic metric is unsupported.")
        return {
            "tag": _validate_optional_tag(config.get("tag"), "Range Statistic tag"),
            "metric": metric,
            "stateValue": _safe_text(config.get("stateValue") or "", "State value", 32, allow_empty=True),
            "units": _safe_text(config.get("units") or "", "Range Statistic units", 32, allow_empty=True),
            "precision": _int(config.get("precision", DEFAULT_AXIS_PRECISION), "Range Statistic precision", 0, 8)
        }

    if widget_type == "stateDuration":
        return {"tag": _validate_optional_tag(config.get("tag"), "State Duration tag"), "states": _validate_state_labels(config.get("states"), "State Duration states")}

    if widget_type == "recentEvents":
        tags = [_validate_tag_config(item, "Recent Events tag") for item in _require_list(config.get("tags", []), "Recent Events tags")]
        if len(tags) > MAX_EVENT_TAGS:
            raise ApiError(400, "Recent Events has too many tags.")
        return {"tags": tags, "maxRows": _int(config.get("maxRows", 12), "Recent Events rows", 1, 100)}

    raise ApiError(400, "Unsupported widget type.")

def _rectangles_overlap(left, right):
    return not (
        left["x"] + left["w"] <= right["x"] or
        right["x"] + right["w"] <= left["x"] or
        left["y"] + left["h"] <= right["y"] or
        right["y"] + right["h"] <= left["y"]
    )

def _validate_dashboard_config(value):
    config = _require_dict(value, "Dashboard config")
    _assert_known_keys(config, ("rows", "cols", "widgets"), "Dashboard config")
    rows = _int(config.get("rows"), "Dashboard rows", GRID_ROWS, GRID_ROWS)
    cols = _int(config.get("cols"), "Dashboard columns", GRID_COLUMNS, GRID_COLUMNS)
    widgets_raw = _require_list(config.get("widgets", []), "Dashboard widgets")
    if len(widgets_raw) > MAX_WIDGETS_PER_DASHBOARD:
        raise ApiError(400, "Dashboard has too many widgets.")

    widgets = []
    seen_ids = set()
    for item in widgets_raw:
        raw = _require_dict(item, "Widget")
        _assert_known_keys(raw, (
            "id", "type", "title", "x", "y", "w", "h", "showHeader",
            "contentScale", "config"
        ), "Widget")
        widget_type = _as_text(raw.get("type"))
        if widget_type not in _ALLOWED_WIDGET_TYPES:
            raise ApiError(400, "Unsupported widget type.")
        widget_id = _safe_text(raw.get("id"), "Widget id", 96, allow_empty=False)
        if not _WIDGET_ID_RE.match(widget_id) or widget_id in seen_ids:
            raise ApiError(400, "Widget IDs must be unique, simple identifiers.")
        seen_ids.add(widget_id)

        min_w, min_h = _MIN_WIDGET_SIZES[widget_type]
        x = _int(raw.get("x"), "Widget x", 0, GRID_COLUMNS - 1)
        y = _int(raw.get("y"), "Widget y", 0, GRID_ROWS - 1)
        w = _int(raw.get("w"), "Widget width", min_w, GRID_COLUMNS)
        h = _int(raw.get("h"), "Widget height", min_h, GRID_ROWS)
        if x + w > GRID_COLUMNS or y + h > GRID_ROWS:
            raise ApiError(400, "Widget bounds must remain inside the 24 x 16 dashboard grid.")

        widgets.append({
            "id": widget_id,
            "type": widget_type,
            "title": _safe_text(raw.get("title") or _default_widget(widget_id, widget_type)["title"], "Widget title", MAX_WIDGET_TITLE_LENGTH, allow_empty=False),
            "x": x,
            "y": y,
            "w": w,
            "h": h,
            "showHeader": _bool(raw.get("showHeader"), True),
            "contentScale": _int(raw.get("contentScale", DEFAULT_WIDGET_CONTENT_SCALE), "Widget content scale", MIN_WIDGET_CONTENT_SCALE, MAX_WIDGET_CONTENT_SCALE),
            "config": _validate_widget_config(widget_type, raw.get("config") or {})
        })

    for index in range(len(widgets)):
        for other_index in range(index + 1, len(widgets)):
            if _rectangles_overlap(widgets[index], widgets[other_index]):
                raise ApiError(400, "Widgets may not overlap. Move or resize one of the tiles.")

    clean = {"rows": rows, "cols": cols, "widgets": widgets}
    if _json_bytes(clean) > MAX_CONFIG_JSON_BYTES:
        raise ApiError(400, "Dashboard configuration is too large.")
    return clean

def _create_dashboard(identity, payload):
    permissions = _require_edit(identity)
    if _count_owner_dashboards(identity["username"]) >= MAX_DASHBOARDS_PER_OWNER:
        raise ApiError(400, "You have reached the configured dashboard limit.")

    payload = _require_dict(payload or {}, "Create dashboard payload")
    _assert_known_keys(payload, ("name", "description"), "Create dashboard payload")
    requested_name = payload.get("name") or DEFAULT_DASHBOARD_NAME
    dashboard_name = _unique_dashboard_name(identity["username"], requested_name)
    description = _safe_text(payload.get("description") or DEFAULT_DASHBOARD_DESCRIPTION, "Dashboard description", MAX_DASHBOARD_DESCRIPTION_LENGTH, allow_empty=True)
    dashboard_id = _as_text(UUID.randomUUID())
    config = _default_dashboard_config()
    config_json = system.util.jsonEncode(config)

    sql = (
        "INSERT INTO %s ([DashboardId], [OwnerUsername], [OwnerDisplayName], "
        "[DashboardName], [DashboardDescription], [IsShared], [ConfigSchemaVersion], [ConfigJson]) "
        "VALUES (CONVERT(UNIQUEIDENTIFIER, ?), ?, ?, ?, ?, ?, ?, ?)" % _DASHBOARD_TABLE_SQL
    )
    _update(sql, [
        dashboard_id, identity["username"], identity["displayName"], dashboard_name,
        description, 1 if DEFAULT_NEW_DASHBOARD_SHARED else 0,
        CONFIG_SCHEMA_VERSION, config_json
    ])
    row, saved_config = _find_dashboard(dashboard_id)
    return _dashboard_payload(row, saved_config, identity, permissions)

def _save_dashboard(identity, payload):
    permissions = _require_edit(identity)
    payload = _require_dict(payload, "Save dashboard payload")
    _assert_known_keys(payload, (
        "dashboardId", "rowVersion", "name", "description", "isShared", "config"
    ), "Save dashboard payload")

    dashboard_id = _safe_text(payload.get("dashboardId"), "Dashboard ID", 64, allow_empty=False)
    row_version = _safe_text(payload.get("rowVersion"), "Dashboard row version", 34, allow_empty=False)
    row, _stored_config = _find_dashboard(dashboard_id)
    access = _dashboard_access(row, identity, permissions)
    if not access["canEdit"]:
        raise ApiError(403, "Only the owner may edit this original dashboard.")

    name = _safe_text(payload.get("name"), "Dashboard name", MAX_DASHBOARD_NAME_LENGTH, allow_empty=False)
    description = _safe_text(payload.get("description") or "", "Dashboard description", MAX_DASHBOARD_DESCRIPTION_LENGTH, allow_empty=True)
    is_shared = _bool(payload.get("isShared"), False)
    if is_shared and not ALLOW_SHARED_DASHBOARDS:
        raise ApiError(403, "Dashboard sharing is disabled by configuration.")
    config = _validate_dashboard_config(payload.get("config"))
    config_json = system.util.jsonEncode(config)

    # The RowVersion predicate prevents silent overwrite when another browser
    # saved first. It is parameterized; browser values never touch SQL syntax.
    sql = (
        "UPDATE %s SET [DashboardName] = ?, [DashboardDescription] = ?, "
        "[IsShared] = ?, [ConfigSchemaVersion] = ?, [ConfigJson] = ?, "
        "[UpdatedAt] = SYSUTCDATETIME() "
        "WHERE [DashboardId] = CONVERT(UNIQUEIDENTIFIER, ?) "
        "AND [RowVersion] = CONVERT(VARBINARY(8), ?, 1)" % _DASHBOARD_TABLE_SQL
    )
    changed = _update(sql, [
        name, description, 1 if is_shared else 0, CONFIG_SCHEMA_VERSION,
        config_json, dashboard_id, row_version
    ])
    if int(changed) != 1:
        raise ApiError(409, "This dashboard was saved in another session. Reload it before saving your changes.")

    saved_row, saved_config = _find_dashboard(dashboard_id)
    return _dashboard_payload(saved_row, saved_config, identity, permissions)

def _delete_dashboard(identity, payload):
    permissions = _require_edit(identity)
    payload = _require_dict(payload, "Delete dashboard payload")
    _assert_known_keys(payload, ("dashboardId", "rowVersion"), "Delete dashboard payload")
    dashboard_id = _safe_text(payload.get("dashboardId"), "Dashboard ID", 64, allow_empty=False)
    row_version = _safe_text(payload.get("rowVersion"), "Dashboard row version", 34, allow_empty=False)
    row, _config = _find_dashboard(dashboard_id)
    access = _dashboard_access(row, identity, permissions)
    if not access["canEdit"]:
        raise ApiError(403, "Only the owner may delete this original dashboard.")

    sql = (
        "DELETE FROM %s WHERE [DashboardId] = CONVERT(UNIQUEIDENTIFIER, ?) "
        "AND [RowVersion] = CONVERT(VARBINARY(8), ?, 1)" % _DASHBOARD_TABLE_SQL
    )
    changed = _update(sql, [dashboard_id, row_version])
    if int(changed) != 1:
        raise ApiError(409, "This dashboard changed in another session. Reload before deleting.")
    return {"dashboardId": dashboard_id}

def _copy_shared_dashboard(identity, payload):
    permissions = _require_edit(identity)
    if not ALLOW_SHARED_DASHBOARD_COPY:
        raise ApiError(403, "Copying shared dashboards is disabled by configuration.")
    if _count_owner_dashboards(identity["username"]) >= MAX_DASHBOARDS_PER_OWNER:
        raise ApiError(400, "You have reached the configured dashboard limit.")

    payload = _require_dict(payload, "Copy dashboard payload")
    _assert_known_keys(payload, ("dashboardId",), "Copy dashboard payload")
    source_id = _safe_text(payload.get("dashboardId"), "Dashboard ID", 64, allow_empty=False)
    source_row, source_config = _find_dashboard(source_id)
    access = _dashboard_access(source_row, identity, permissions)
    if not access["canRead"]:
        raise ApiError(403, "You are not authorized to copy this dashboard.")
    if not access["isOwner"] and not bool(source_row["IsShared"]):
        raise ApiError(403, "Only shared dashboards may be copied by another user.")

    config = _validate_dashboard_config(system.util.jsonDecode(system.util.jsonEncode(source_config)))
    new_id = _as_text(UUID.randomUUID())
    source_name = _as_text(source_row["DashboardName"])
    new_name = _unique_dashboard_name(identity["username"], source_name + " (Copy)")
    description = _safe_text(source_row["DashboardDescription"] or "", "Dashboard description", MAX_DASHBOARD_DESCRIPTION_LENGTH, allow_empty=True)

    sql = (
        "INSERT INTO %s ([DashboardId], [OwnerUsername], [OwnerDisplayName], "
        "[DashboardName], [DashboardDescription], [IsShared], [ConfigSchemaVersion], [ConfigJson]) "
        "VALUES (CONVERT(UNIQUEIDENTIFIER, ?), ?, ?, ?, ?, 0, ?, ?)" % _DASHBOARD_TABLE_SQL
    )
    _update(sql, [
        new_id, identity["username"], identity["displayName"], new_name, description,
        CONFIG_SCHEMA_VERSION, system.util.jsonEncode(config)
    ])
    row, saved_config = _find_dashboard(new_id)
    return _dashboard_payload(row, saved_config, identity, permissions)

def _list_shared(identity):
    permissions = _require_view(identity)
    if not ALLOW_SHARED_DASHBOARDS:
        return {"dashboards": []}

def _safe_value(value):
    if value is None:
        return None
    # Java Date is returned as epoch milliseconds.
    try:
        if hasattr(value, "getTime"):
            return long(value.getTime())
    except Exception:
        pass
    # Simple scalars remain JSON numbers/bools/strings.
    if isinstance(value, (bool, int, long, float, unicode, str)):
        return value
    try:
        return _as_text(value)
    except Exception:
        return None

def _quality_text(quality):
    try:
        return _as_text(quality)
    except Exception:
        return ""

def _browse_tags(identity, payload):
    _require_view(identity)
    payload = _require_dict(payload or {}, "Tag browse payload")
    _assert_known_keys(payload, ("path", "filter"), "Tag browse payload")

    requested_path = payload.get("path")
    if not requested_path:
        roots = []
        for item in ALLOWED_TAG_ROOTS:
            roots.append({
                "path": item["root"],
                "name": item.get("label") or item["root"],
                "dataType": "",
                "tagType": "Folder",
                "hasChildren": True,
                "historyEligible": False,
                "isRoot": True
            })
        return {"parentPath": None, "nodes": roots}

    parent_path = _validate_tag_path(requested_path, "Browse path")
    root = _allowed_root_for_path(parent_path)
    if root is None:
        raise ApiError(403, "Browse path is outside the approved roots.")
    if _path_depth(_as_text(root["root"]), parent_path) > TAG_BROWSER_MAX_DEPTH:
        raise ApiError(400, "The configured tag browser depth has been reached.")

    filter_text = _safe_text(payload.get("filter") or "", "Tag filter", 100, allow_empty=True)
    browse_filter = {"maxResults": TAG_BROWSER_MAX_RESULTS}
    if filter_text:
        browse_filter["name"] = "*" + filter_text + "*"

    try:
        raw_nodes = system.tag.browse(parent_path, browse_filter).getResults()
    except Exception as exc:
        _log_exception("Tag browse failed", exc)
        raise ApiError(500, "Unable to browse the configured tag root.")

    nodes = []
    for node in raw_nodes:
        full_path = _as_text(node.get("fullPath"))
        if _allowed_root_for_path(full_path) is None:
            continue
        has_children = bool(node.get("hasChildren"))
        tag_type = _as_text(node.get("tagType") or "")
        data_type = _as_text(node.get("dataType") or "")
        is_atomic = tag_type == "AtomicTag" or (not has_children and data_type)
        history_eligible = False
        if is_atomic:
            history_eligible = _history_enabled_for_tag(full_path)
        nodes.append({
            "path": full_path,
            "name": _as_text(node.get("name") or full_path),
            "dataType": data_type,
            "tagType": tag_type,
            "hasChildren": has_children,
            "historyEligible": history_eligible,
            "kind": _tag_kind_from_datatype(data_type)
        })
    return {"parentPath": parent_path, "nodes": nodes}

def _get_tag_metadata(identity, payload):
    _require_view(identity)
    payload = _require_dict(payload, "Tag metadata payload")
    _assert_known_keys(payload, ("path",), "Tag metadata payload")
    meta = _tag_metadata(payload.get("path"))
    # Normalize server metadata to the persisted tag-instance contract. The
    # browser still treats it as advisory; _validate_tag_config revalidates
    # only the selected path and safe configuration fields during save.
    meta["displayName"] = meta.get("name") or meta.get("path")
    meta["datatype"] = meta.get("dataType") or ""
    meta["suggestedRenderMode"] = meta.get("kind") or "analog"
    if REQUIRE_HISTORY_ENABLED_TAGS and not meta["historyEligible"]:
        meta["selectable"] = False
    else:
        meta["selectable"] = True
    return {"tag": meta}

def _enforce_history_rate(identity):
    now = long(system.date.now().getTime())
    cutoff = now - 60000
    username = identity["username"]
    with _HISTORY_RATE_LOCK:
        bucket = _HISTORY_RATE_BUCKETS.get(username, [])
        bucket = [timestamp for timestamp in bucket if timestamp >= cutoff]
        if len(bucket) >= MAX_HISTORY_REQUESTS_PER_MINUTE_PER_USER:
            _HISTORY_RATE_BUCKETS[username] = bucket
            raise ApiError(429, "Historian request rate limit reached. Wait a moment before requesting more data.")
        bucket.append(now)
        _HISTORY_RATE_BUCKETS[username] = bucket

def _parse_time_ms(value, label):
    timestamp = _int(value, label, 0)
    # Reject clearly invalid dates and accidental seconds-vs-milliseconds input.
    if timestamp < 946684800000:
        raise ApiError(400, "%s must be epoch milliseconds." % label)
    return timestamp

def _history_request(identity, payload):
    _require_view(identity)
    _enforce_history_rate(identity)
    payload = _require_dict(payload, "History payload")
    _assert_known_keys(payload, (
        "paths", "startMs", "endMs", "returnSize", "aggregationModes"
    ), "History payload")

    paths = _validate_tag_paths(payload.get("paths"), MAX_HISTORY_TAGS_PER_REQUEST, "History paths")
    start_ms = _parse_time_ms(payload.get("startMs"), "History start")
    end_ms = _parse_time_ms(payload.get("endMs"), "History end")
    if end_ms <= start_ms:
        raise ApiError(400, "History end must be after history start.")
    duration_seconds = (end_ms - start_ms) / 1000.0
    if duration_seconds > MAX_HISTORY_DURATION_SECONDS:
        raise ApiError(400, "Requested history range exceeds the configured maximum.")

    return_size = _int(payload.get("returnSize", DEFAULT_HISTORY_RETURN_SIZE), "History return size", -1, MAX_HISTORY_POINTS_PER_TAG)
    if return_size == 0:
        raise ApiError(400, "Natural historian return size is disabled for predictable request limits.")

    aggregation_modes = _require_list(payload.get("aggregationModes", []), "History aggregation modes")
    if aggregation_modes and len(aggregation_modes) != len(paths):
        raise ApiError(400, "History aggregation modes must match the requested paths.")
    if not aggregation_modes:
        aggregation_modes = ["Raw" if return_size == -1 else DEFAULT_AGGREGATION_MODE for _ in paths]

    raw_requested = return_size == -1
    if raw_requested:
        if not USE_RAW_HISTORY_BY_DEFAULT:
            raise ApiError(403, "Raw historian retrieval is disabled by configuration.")
        if duration_seconds > MAX_RAW_HISTORY_DURATION_SECONDS:
            raise ApiError(400, "Raw history is limited to the configured shorter range. Request an approved aggregation for a larger range.")
        for mode in aggregation_modes:
            if _as_text(mode) != "Raw":
                raise ApiError(400, "Raw history requests must use Raw for every selected tag.")
    else:
        for mode in aggregation_modes:
            if _as_text(mode) not in ALLOWED_HISTORY_AGGREGATION_MODES:
                raise ApiError(400, "Unsupported history aggregation mode.")

    start_date = Date(start_ms)
    end_date = Date(end_ms)
    try:
        if raw_requested:
            dataset = system.tag.queryTagHistory(
                paths=paths,
                startDate=start_date,
                endDate=end_date,
                returnSize=-1,
                returnFormat="Wide",
                includeBoundingValues=False,
                noInterpolation=True,
                ignoreBadQuality=True
            )
        else:
            dataset = system.tag.queryTagHistory(
                paths=paths,
                startDate=start_date,
                endDate=end_date,
                returnSize=return_size,
                aggregationModes=aggregation_modes,
                returnFormat="Wide",
                includeBoundingValues=True,
                noInterpolation=False,
                ignoreBadQuality=True
            )
    except Exception as exc:
        _log_exception("Historian query failed", exc)
        raise ApiError(500, "The historian query failed. Verify tag history and gateway historian availability.")

    # Convert the wide Ignition dataset into browser-friendly per-tag series.
    pydata = system.dataset.toPyDataSet(dataset)
    series = [{"path": path, "points": []} for path in paths]
    row_count = 0
    for row in pydata:
        row_count += 1
        timestamp = _millis(row[0])
        if timestamp is None:
            continue
        for index in range(len(paths)):
            value = _safe_value(row[index + 1])
            if value is not None:
                series[index]["points"].append([timestamp, value])

    # The server cannot reliably predict raw on-change density for every
    # historian implementation. Stop returning oversized raw replies even if a
    # misconfigured high-frequency tag was allowed through the duration guard.
    if raw_requested and row_count > MAX_HISTORY_POINTS_PER_TAG:
        raise ApiError(413, "Raw history returned more points than allowed. Use an approved aggregation or reduce the time window.")

    return {
        "paths": paths,
        "series": series,
        "startMs": start_ms,
        "endMs": end_ms,
        "returnSize": return_size,
        "aggregationModes": aggregation_modes,
        "rowCount": row_count
    }

def _current_request(identity, payload):
    _require_view(identity)
    payload = _require_dict(payload, "Current value payload")
    _assert_known_keys(payload, ("paths",), "Current value payload")
    paths = _validate_tag_paths(payload.get("paths"), MAX_CURRENT_TAGS_PER_REQUEST, "Current paths")
    try:
        values = system.tag.readBlocking(paths)
    except Exception as exc:
        _log_exception("Current tag batch read failed", exc)
        raise ApiError(500, "Current tag read failed.")
    result = []
    for index, path in enumerate(paths):
        qualified = values[index]
        result.append({
            "path": path,
            "value": _safe_value(qualified.value),
            "quality": _quality_text(qualified.quality),
            "timestamp": _millis(qualified.timestamp)
        })
    return {"values": result}

def _as_of_request(identity, payload):
    _require_view(identity)
    _enforce_history_rate(identity)
    payload = _require_dict(payload, "As-of payload")
    _assert_known_keys(payload, ("paths", "focusMs"), "As-of payload")
    paths = _validate_tag_paths(payload.get("paths"), MAX_CURRENT_TAGS_PER_REQUEST, "As-of paths")
    focus_ms = _parse_time_ms(payload.get("focusMs"), "Reference time")
    start_ms = focus_ms - (AS_OF_LOOKBACK_SECONDS * 1000)
    try:
        dataset = system.tag.queryTagHistory(
            paths=paths,
            startDate=Date(start_ms),
            endDate=Date(focus_ms),
            returnSize=-1,
            returnFormat="Wide",
            includeBoundingValues=INCLUDE_BOUNDING_VALUES_FOR_AS_OF_LOOKUPS,
            noInterpolation=True,
            ignoreBadQuality=True
        )
    except Exception as exc:
        _log_exception("As-of historian query failed", exc)
        raise ApiError(500, "The historian as-of query failed.")

    latest = {}
    pydata = system.dataset.toPyDataSet(dataset)
    for row in pydata:
        timestamp = _millis(row[0])
        if timestamp is None or timestamp > focus_ms:
            continue
        for index, path in enumerate(paths):
            value = _safe_value(row[index + 1])
            if value is not None:
                latest[path] = {"path": path, "value": value, "quality": "Good", "timestamp": timestamp}

    result = []
    for path in paths:
        if path in latest:
            result.append(latest[path])
        else:
            result.append({"path": path, "value": None, "quality": "No data", "timestamp": None})
    return {"focusMs": focus_ms, "values": result}

def _bootstrap(identity, session):
    permissions = _require_view(identity)
    permissions["canAdmin"] = permissions.get("isAdmin", False)
    if not session.get("pmdCsrf"):
        session["pmdCsrf"] = _as_text(UUID.randomUUID())
    return {
        "appTitle": APP_TITLE,
        "appVersion": APP_VERSION,
        "username": identity["username"],
        "displayName": identity["displayName"],
        "roles": identity["roles"],
        "permissions": permissions,
        "csrfToken": session["pmdCsrf"],
        "limits": {
            "gridColumns": GRID_COLUMNS,
            "gridRows": GRID_ROWS,
            "maxWidgets": MAX_WIDGETS_PER_DASHBOARD,
            "maxTrendPens": MAX_TREND_PENS_PER_WIDGET,
            "maxAxes": MAX_AXES_PER_TREND,
            "maxHistoryDurationSeconds": MAX_HISTORY_DURATION_SECONDS,
            "maxHistoryPointsPerTag": MAX_HISTORY_POINTS_PER_TAG,
            "minPollSeconds": MIN_POLL_SECONDS,
            "maxPollSeconds": MAX_POLL_SECONDS,
            "defaultPollSeconds": DEFAULT_POLL_SECONDS
        }
    }

def _dispatch(request, session, body):
    body = _require_dict(body, "Request")
    _assert_known_keys(body, ("action", "payload"), "Request")
    action = _safe_text(body.get("action"), "Action", 80, allow_empty=False)
    payload = body.get("payload") or {}

    identity = _authenticated_identity(session)
    _ensure_csrf(session, action, request)

    if action == "auth.bootstrap":
        return _bootstrap(identity, session)
    if action == "dashboard.listMine":
        return _list_mine(identity)
    if action == "dashboard.listShared":
        return _list_shared(identity)
    if action == "dashboard.get":
        return _get_dashboard(identity, payload)
    if action == "dashboard.create":
        return {"dashboard": _create_dashboard(identity, payload)}
    if action == "dashboard.save":
        return {"dashboard": _save_dashboard(identity, payload)}
    if action == "dashboard.delete":
        return _delete_dashboard(identity, payload)
    if action == "dashboard.copyShared":
        return {"dashboard": _copy_shared_dashboard(identity, payload)}
    if action == "tag.browse":
        return _browse_tags(identity, payload)
    if action == "tag.getMetadata":
        return _get_tag_metadata(identity, payload)
    if action == "data.readCurrentBatch":
        return _current_request(identity, payload)
    if action == "data.historyBatch":
        return _history_request(identity, payload)
    if action == "data.asOfBatch":
        return _as_of_request(identity, payload)

    raise ApiError(404, "Unknown API action.")

  function durationByState(points,start,end) {
    var totals={}, sorted=points.slice().sort(function(a,b){return a.timestamp-b.timestamp;});
    if(!sorted.length)return totals;
    var cur=sorted[0].value, at=start;
    sorted.forEach(function(p){if(p.timestamp<start)return; var cut=Math.min(end,p.timestamp);if(cut>at){totals[String(cur)]=(totals[String(cur)]||0)+(cut-at);}cur=p.value;at=Math.max(at,p.timestamp);});
    if(end>at)totals[String(cur)]=(totals[String(cur)]||0)+(end-at);
    return totals;
  }
  function durationText(ms) {
    ms=Math.max(0,ms||0);var sec=Math.floor(ms/1000), h=Math.floor(sec/3600),m=Math.floor((sec%3600)/60),s=sec%60;return (h?h+"h ":"")+(m?m+"m ":"")+s+"s";
  }

  function startLayoutPointer(ev,w,mode) {
    if(state.mode!=="edit" || !canEditCurrent())return;
    ev.preventDefault(); ev.stopPropagation();
    var canvas=$("dashboardCanvas"), rect=canvas.getBoundingClientRect(), origin={x:w.x,y:w.y,ww:w.w,hh:w.h,px:ev.clientX,py:ev.clientY};
    canvas.classList.add("dragging"); ev.target.setPointerCapture && ev.target.setPointerCapture(ev.pointerId);
    function move(e) {
      var dx=Math.round((e.clientX-origin.px)/rect.width*C.grid.columns), dy=Math.round((e.clientY-origin.py)/rect.height*C.grid.rows);
      var next=clone(w);
      if(mode==="drag"){next.x=Math.max(0,Math.min(C.grid.columns-next.w,origin.x+dx));next.y=Math.max(0,Math.min(C.grid.rows-next.h,origin.y+dy));}
      else {var meta=WIDGET_META[w.type];next.w=Math.max(meta.minW,Math.min(C.grid.columns-origin.x,origin.ww+dx));next.h=Math.max(meta.minH,Math.min(C.grid.rows-origin.y,origin.hh+dy));}
      if(!collides(next,w.id)){w.x=next.x;w.y=next.y;w.w=next.w;w.h=next.h;var tile=canvas.querySelector('[data-widget-id="'+w.id+'"]');if(tile)tile.style.cssText=gridStyle(w);setDirty(true);}
    }
    function up(){canvas.classList.remove("dragging");window.removeEventListener("pointermove",move);window.removeEventListener("pointerup",up);renderCanvas();}
    window.addEventListener("pointermove",move);window.addEventListener("pointerup",up);
  }
  function collides(candidate,ignoreId) {

  }

def doGet(request, session):
    try:
        _enforce_transport(request)
        identity = _authenticated_identity(session)
        _require_view(identity)
        return {"html": _build_page_html()}
    except ApiError as exc:
        _set_http_status(request, exc.status)
        return {"html": _error_page_html(exc.status, exc.message)}
    except Exception as exc:
        _log_exception("doGet failed", exc)
        _set_http_status(request, 500)
        return {"html": _error_page_html(500, "The dashboard could not be opened. Check Gateway logs.")}

def doPost(request, session):
    try:
        _enforce_transport(request)
        body = request.get("data")
        if body is None:
            body = request.get("postData")
        if _json_bytes(body) > MAX_JSON_REQUEST_BYTES:
            raise ApiError(413, "Request is too large.")
        # Depending on gateway/resource version, JSON may be provided as a
        # decoded map or as raw text. Support both without accepting form data.
        if not isinstance(body, dict):
            try:
                body = system.util.jsonDecode(_as_text(body))
            except Exception:
                pass
        result = _dispatch(request, session, body)
        return _json_ok(request, result)
    except ApiError as exc:
        return _json_error(request, exc.status, exc.message)
    except Exception as exc:
        request_id = _as_text(UUID.randomUUID())
        _log_exception("doPost failed [%s]" % request_id, exc)
        return _json_error(request, 500, "The server could not complete that request. Check Gateway logs with request ID %s." % request_id, request_id)
